-- SOAL 01-5: CARI NILAI TERBESAR DARI TIGA ANGKA --
-- Diketahui: a = 7, b = 12, c = 4
-- Target output: 12
--
-- Tulis pseudocode-mu di bawah baris ini:

start

    store a with value 7
    store b with value 12
    store c with value 9
    store terbesar without any value
   

    if nilai a > b and a > c then set terbesar "a"
    else if nilai b > a and b > c then set terbesar "b"
    else set terbesar "c"

    display "nilai terbesar : " + terbesar

end
